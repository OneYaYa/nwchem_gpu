cmake_minimum_required(VERSION 3.24)
project(nwchem_gpu VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(NWGPU_ENABLE_CUDA "Build the NVIDIA CUDA backend (requires nvcc)" OFF)
option(NWGPU_USE_BLAS "Use system LP64 BLAS for CPU GEMM" ON)
set(NWGPU_FORTRAN_INTEGER_BYTES 8 CACHE STRING "NWChem INTEGER/LOGICAL width: 4 or 8")
if(NOT NWGPU_FORTRAN_INTEGER_BYTES MATCHES "^(4|8)$")
  message(FATAL_ERROR "NWGPU_FORTRAN_INTEGER_BYTES must be 4 or 8")
endif()
add_library(nwchem_gpu SHARED src/backend.cpp src/fortran.cpp)
target_compile_features(nwchem_gpu PUBLIC cxx_std_17)
target_include_directories(nwchem_gpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(nwchem_gpu PUBLIC NWGPU_FORTRAN_INTEGER_BYTES=${NWGPU_FORTRAN_INTEGER_BYTES})
set_target_properties(nwchem_gpu PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(NWGPU_USE_BLAS)
  set(BLA_SIZEOF_INTEGER 4)
  find_package(BLAS REQUIRED)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_LIBRARIES BLAS::BLAS)
  check_cxx_source_compiles("extern \"C\" void cblas_dgemm(); int main() { cblas_dgemm(); }" NWGPU_HAS_CBLAS)
  unset(CMAKE_REQUIRED_LIBRARIES)
  if(NOT NWGPU_HAS_CBLAS)
    message(FATAL_ERROR "LP64 CBLAS is required: use OpenBLAS, MKL LP64, Accelerate, or NWGPU_USE_BLAS=OFF")
  endif()
  target_link_libraries(nwchem_gpu PRIVATE BLAS::BLAS)
  target_compile_definitions(nwchem_gpu PRIVATE NWGPU_HAVE_BLAS=1)
endif()
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/nwchem_gpu_config.json
  CONTENT "{\"fortran_integer_bytes\":${NWGPU_FORTRAN_INTEGER_BYTES},\"cuda\":$<IF:$<BOOL:${NWGPU_ENABLE_CUDA}>,true,false>,\"system_blas\":$<IF:$<BOOL:${NWGPU_USE_BLAS}>,true,false>}\n")
if(NWGPU_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(nwchem_gpu PRIVATE src/cuda_backend.cu)
  target_compile_definitions(nwchem_gpu PRIVATE NWGPU_HAVE_CUDA=1)
  target_link_libraries(nwchem_gpu PRIVATE CUDA::cudart CUDA::cublas)
  set_target_properties(nwchem_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
add_executable(nwgpu_bench src/benchmark.cpp)
target_link_libraries(nwgpu_bench PRIVATE nwchem_gpu)
if(BUILD_TESTING)
  add_executable(nwgpu_tests tests/test_backend.cpp)
  target_link_libraries(nwgpu_tests PRIVATE nwchem_gpu)
  add_test(NAME cpu_contract COMMAND nwgpu_tests cpu)
  if(NWGPU_ENABLE_CUDA)
    add_test(NAME cuda_contract COMMAND nwgpu_tests cuda)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME integration_overlay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_prepare.py)
  endif()
endif()
install(TARGETS nwchem_gpu nwgpu_bench LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/nwchem_gpu.h DESTINATION include)
